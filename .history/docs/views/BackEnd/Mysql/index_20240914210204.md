## DDL
1. 创建表
```sql

```

```
CREATE TABLE  表明 ( 属性1 数据类型1, 属性2 数据类型2, ...) comment '注释';
```

|     属性名     |   数据类型   | 默认值 | 是否为空 |   注释   |
| :------------: | :----------: | :----: | :------: | :------: |
|  primary key   |   int(11)    |  NULL  | NOT NULL |   主键   |

|     index      | varchar(255) |  NULL  | NOT NULL | 普通索引 |
| auto_increment |   int(11)    |  NULL  | NOT NULL | 自增索引 |

|    not null    | varchar(255) |  NULL  | NOT NULL |          |

# DQL 查询数据

::: 语法
1. 查询多个字段 select 列名1,列名2,列名3 from 表名;
2. 查询所有字敦(通配符): select * from 表名;
3. 设置路径别名： select 列名 as 别名 from 表名;
4. 模糊查询 select 列名 like '值%';
5. 分页查询： select * from 表名 limit 开始位置,每页显示条数;
6. 去除重复数据 select distinct 列名 from 表名;
:::

::: TIP
SELECT 字段列表
FROM 表名
[WHERE 条件] 
[GROUP BY 分组字段] 
[HAVING 分组条件]
[ORDER BY 排序字段] 
[LIMIT 开始位置,每页显示条数]
[between 值1 and 值2] //之间

select * from user where id > 5;
select * from user where id > 5 and username = 'admin';
<!-- 逻辑运算符  and or -->
select * from user where id > 5 or username = 'admin';
<!-- 逻辑运算符  and or  not  -->
select * from user where id > 5 and username = 'admin' or username = 'admin';
<!-- 括号 优先级最高  -->
select * from user where id > 5 and (username = 'admin' or username = 'admin')
<!-- 模糊查询 查询user 表中 id 大于 5 且 username 中包含 admin -->
select * from user where id > 5 and username like '%admin%';
<!-- 模糊查询 查询user 表中 id 大于 5 且 username 以 admin 开头 -->
select * from user where id > 5 and username like 'admin%';
select * from user where id > 5 and username like '%admin';
:::

## count 统计数量

```
select count(id) from user;
<!-- 统计表中名称不重复的个数 -->
select count(distinct username) from user;
<!-- 统计表中所有数据的个数 -->
select count(*) from user;


```

## min 最小值
```
<!-- 获取用户表中年龄最小的  -->
select min(age) from user;
```
## max 最大值
```
<!-- 统计表中年龄最大的 -->
select max(age) from user;
```

## avg 平均值

```
<!--获取平局值 -->
select avg(age) from user;

```

##  sum 求和

```
<!-- 获取和 -->
select sum(age) from user;
```

##  分组查询

```
<!-- 分组查询 age 统计用户表中 每个年龄的个数 -->
select age,count(*) from user group by age;
select age,count(*) from user group by age having count(*) > 1;
```

## 排序 order by

```
<!-- 排序 age 倒序  -->
select * from user order by age desc;
<!-- 排序 age 升序 -->
select * from user order by age asc;
<!-- 分页 排序 age 倒序 -->
select * from user order by age desc limit 2;
select * from user order by age desc limit 2 offset 2;
select * from user order by age desc limit 2,2; 

select * from user order by age desc, updated_at asc;
```

## rand() 随机

```
select * from user order by rand() limit 2;
select * from user order by rand() limit 2 offset 2;
select * from user order by rand() limit 2,2;
```

## in
```
<!-- in 查询 -id为1、2或3的记录。->
select * from user where id in (1,2,3);
select * from user where id not in (1,2,3);
```

## having 分组后过滤
```
select age,count(*) from user group by age having count(*) > 1;
```

## limit 分页
offset 从第几条开始
```
select 字段列表 from 表名 limit 开始索引,每页数目;
<!-- 分页 查询 user 表中每页显示2条数据 -->
select * from user limit 2;

select * from user limit 2 offset 2;
<!-- 查询 user 表中每页显示2条数据，从第3条开始 -->
select * from user limit 2,2;

select * from user limit 2 offset 2;
```
